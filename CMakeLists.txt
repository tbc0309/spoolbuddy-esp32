cmake_minimum_required(VERSION 3.22)
set(SB_ROOT "${CMAKE_CURRENT_LIST_DIR}")
set(SB_BOARD "esp32-p4-wifi6-touch-lcd-4.3/rev3" CACHE STRING "Board directory under boards/")
if(NOT SB_BOARD MATCHES "^[a-zA-Z0-9_./-]+$" OR SB_BOARD MATCHES "\\.\\.")
    message(FATAL_ERROR "Invalid SB_BOARD")
endif()
set(SB_BOARD_DIR "${SB_ROOT}/boards/${SB_BOARD}")
if(NOT EXISTS "${SB_BOARD_DIR}/board.cmake")
    message(FATAL_ERROR "Unsupported board: ${SB_BOARD}; provide a board.cmake profile")
endif()
include("${SB_BOARD_DIR}/board.cmake")
if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL SB_BOARD_TARGET)
    message(FATAL_ERROR "Selected board requires ${SB_BOARD_TARGET}, got ${IDF_TARGET}")
endif()
set(IDF_TARGET "${SB_BOARD_TARGET}" CACHE STRING "" FORCE)
list(APPEND EXTRA_COMPONENT_DIRS "${SB_BOARD_COMPONENT_DIR}")
if(NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "${SB_ROOT}/sdkconfig.defaults;${SB_BOARD_DEFAULTS}")
endif()
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
endif()
if(EXISTS "${CMAKE_BINARY_DIR}/sb-board-id.txt")
    file(READ "${CMAKE_BINARY_DIR}/sb-board-id.txt" SB_PREVIOUS_BOARD)
    if(NOT SB_PREVIOUS_BOARD STREQUAL SB_BOARD)
        message(FATAL_ERROR "Use a fresh build directory when changing SB_BOARD")
    endif()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/sb-board-id.txt" "${SB_BOARD}")
option(SB_UI_PREVIEW "Appearance-only preview" OFF)
include("${CMAKE_CURRENT_LIST_DIR}/cmake/version.cmake")
set(PROJECT_VER "${SB_PORT_VERSION}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(spoolbuddy_port VERSION ${SB_PORT_VERSION})
if(DEFINED SB_BOARD_POST_PROJECT)
    include("${SB_BOARD_POST_PROJECT}")
endif()
if(TARGET idf::espressif__esp_hosted)
    execute_process(COMMAND "${PYTHON}" "${SB_ROOT}/patches/esp_hosted_rx_retry/apply.py"
        "${SB_ROOT}/managed_components/espressif__esp_hosted"
        RESULT_VARIABLE SB_HOSTED_PATCH_RESULT)
    if(NOT SB_HOSTED_PATCH_RESULT EQUAL 0)
        message(FATAL_ERROR "ESP-Hosted RX retry dependency fix could not be applied")
    endif()
endif()
